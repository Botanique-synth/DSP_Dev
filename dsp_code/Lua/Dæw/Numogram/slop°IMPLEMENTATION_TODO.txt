PROJECT OVERVIEW:
================================================================================

This is a Faust DSP project building an Elektron-inspired machine that blends:
- Drum synthesis (kick, snare, hats, percussive elements)
- Synth machines (FM, subtractive, granular, wavetable)
- Samplers (for one-shot and looped samples)
- A sophisticated 16-step sequencer with numogram-based parameter mapping
- Mixer with effects (delay, reverb, EQ)

The core concept uses NUMOGRAM DIAGRAM geometry to organize and relate:
- Machine types to numbers (0=clock, 8=drums, 1=synth, 7=sampler, etc.)
- Step positions to harmonic relationships
- Parameter values to numogram ratios

LUA INTEGRATION: main.lua provides visual interface with numogram display
FAUST ENGINE: Main.dsp contains all DSP processing

================================================================================
CURRENT ARCHITECTURE (from Main.dsp):
================================================================================

process = Plex <: Timecircuit : Warp;

Component Structure:
├── Plex (zero + nine)
│   ├── zero: Clock system - BPM, run control, phasor generation
│   └── nine: 16-step sequencer with numogram-based note generation
│
├── Timecircuit (eight + 3x synth_fx)
│   ├── eight: Drum synthesis (khique, snr, Hate, additive)
│   ├── synth_fx: Synth machine 1 (STUB - needs implementation)
│   ├── synth_fx: Synth machine 2 (STUB - needs implementation)
│   └── synth_fx: Synth machine 3 (STUB - needs implementation)
│
└── Warp (one): 6-channel mixer with effects
    ├── 4 input channels (from drums + synths)
    ├── Delay effect with dispersion
    ├── Reverb (stereo freeverb)
    ├── Per-channel EQ (low/high shelf, bell, passes)
    └── Per-channel volume, delay send, reverb send

// KHIQUE (Kick Drum) - From Main.dsp
aa_khique(trig) = out
with{
    len = hslider("hlen [style:knob]",0.5,0,1,0.001);
    pitch = hslider("hpit[style:knob]",0,0,1,0.001);
    base = hslider("Bf[style:knob]",0,0,1,0.001);
    sat = hslider("sat[style:knob]",0,0,1,0.001);
    mod = hslider("mod[style:knob]",0,0,1,0.001);
    decay_env(decay,trig) = trig : ba.line((10+50000*decay)*(1-trig));
    upper = 200*(1+base); lower = 50*(1+base);
    env = decay_env(len,trig);
    f = decay_env(pitch/10,trig) : lower+(_*(upper-lower));
    osc = os.CZhalfSine(os.sawNp(1,f/2,trig*2*ma.PI),mod)*env;
    out = osc*(1+(5*sat)-env/2) : aa.tanh1;
};

// SNARE - From Main.dsp
aa_snr(trig) = out
with{
    f = hslider("freq[style:knob]",200,200,500,1):si.smoo;
    dec = hslider("decay[style:knob]",0,0,1,0.001);
    n = hslider("noise[style:knob]",0,0,1,0.001);
    env = en.are(0,dec,trig:ba.impulsify:ba.spulse(10));
    out = env*(os.oscsin(f*(1+env))) + 5*env*env*n*no.noise/20;
};

// HATE (Hi-Hat) - From Main.dsp
aa_hate(trig) = out
with{
    fh = hslider("freq_hat[style:knob]",1500,1500,2000,1):si.smoo;
    spreadh = hslider("spread hat[style:knob]",0,0,1,0.001);
    dech = hslider("decay[style:knob]",0,0,1,0.001);
    nh = hslider("noise h[style:knob]",0,0,1,0.001);
    env = en.are(0,dech,trig:ba.impulsify:ba.spulse(10));
    s = 1.2 + spreadh/9;
    summ = env*(os.square(fh) + os.square(fh*s) + os.square(fh*s*s) 
           + os.square(fh*s*s*s) + os.square(fh*s*s*s*s))/6;
    noise = 5*env*nh*no.noise/20;
    out = (summ+noise) : fi.highpass(8,5000+fh/2);
};

// ADDITIVE DRUM - From Main.dsp
aa_additive(trig) = out
with{
    f = hslider("freq[style:knob]",1,1,800,1):si.smoo;
    spread = hslider("spread[style:knob]",0,0,1,0.001);
    dec = (hslider("dec[style:knob]",0.5,0,1,0.01)+0.5)*8;
    env = en.are(0.02,dec,trig:ba.spulse(100));
    s = 1 + spread*3;
    a = os.oscp(f,-dec); b = os.oscp(f*s,dec);
    c = os.oscp(f*s*s,-dec); d = os.oscp(f*s*s*s,dec);
    e = os.oscp(f*s*s*s*s,-dec); g = os.oscp(f*s*s*s*s*s,dec);
    out = env^4*(a+b+c+d+e+g)/6;
};

// SIMPLE SUBTRACTIVE SYNTH (for synth_fx)
aa_simple_synth(trig, freq_param, vel_param, ctrl_param) = out
with{
    note = 60 + (freq_param * 48); // 0-1 -> 60-108 (C4-C7)
    freq = 440 * ma.pow(2, (note-69)/12);
    saw = os.saw(freq);
    square = os.square(freq*1.5);
    waveform = ba.lininterp(ctrl_param, saw, square);
    env = en.adsr(0.001, 0.1, 0.3, 0.1, trig);
    cutoff = 200 + ctrl_param * 10000;
    filtered = waveform : fi.lowpass(cutoff, 0.5);
    out = filtered * env * vel_param * 0.5, filtered * env * vel_param * 0.5;
};

// FM SYNTH (2-operator)
aa_fm_synth(trig, freq_param, vel_param, ctrl_param) = out
with{
    carrier_freq = 440 * ma.pow(2, (60 + freq_param * 48 - 69)/12);
    mod_freq = carrier_freq * 1.5;
    mod_index = ctrl_param * 10;
    mod = os.sinwave(mod_freq);
    carrier = os.sinwave(carrier_freq + mod * mod_index);
    env = en.adsr(0.01, 0.2, 0.5, 0.1, trig);
    out = carrier * env * vel_param * 0.3 : fi.lowpass(5000, 0.3);
};

// KARPLUS-STRONG
aa_karplus(freq, damp, trig) = out
with{
    maxDElL = 960;
    filter = _ <: _,_' :> /(2);
    delLength = ma.SR/freq;
    impulse = trig : ba.impulsify;
    string = +~(de.fdelay4(maxDElL,delLength-1) : filter : *(damp));
    out = string * impulse * 0.5;
};

// 16-STEP TRIGGER SEQUENCER (optimized)
aa_16step(bpm, run) = out
with{
    phasor = os.hs_phasor(1.0, bpm/60, run);
    pulse = (1-phasor):ba.spulse(1);
    count = ba.counter(pulse) % 16;
    step(i) = checkbox("step_" + i);
    out = par(i, 16, step(i) & (count == i) : ba.impulsify);
};

// NUMOGRAM QUANTIZER (from Main.dsp nine)
aa_tune(bf, scale_typ, fin) = fout
with{
    Bf = 20 * pow(1000, bf);
    Fin = 20 * pow(1000, fin);
    scaleAt(0) = qu.ionian; scaleAt(1) = qu.dorian; scaleAt(2) = qu.phrygian;
    scaleAt(3) = qu.lydian; scaleAt(4) = qu.mixo; scaleAt(5) = qu.eolian;
    scaleAt(6) = qu.locrian; scaleAt(7) = qu.pentanat; scaleAt(8) = qu.kumoi;
    scaleAt(9) = qu.natural; scaleAt(10) = qu.dodeca; scaleAt(11) = qu.dimin;
    scaleAt(12) = qu.penta;
    NSCALES = 13;
    idx = int(scale_typ * (NSCALES-1) + 0.5);
    outs = par(i, NSCALES, Fin : qu.quantize(Bf, scaleAt(i)));
    fout = outs : ba.selectn(NSCALES, idx);
};

// STEREO DELAY (from Main.dsp Warp)
aa_stereo_delay(input_l, input_r, duration, feedback, dfactor) = outl, outr
with{
    max_del = 15*ma.SR;
    Dl = duration*ma.SR; Dr = duration*ma.SR*1.5:si.smooth(.9);
    wow = 1 + os.sinwaveform(600000)*os.sinwaveform(700201):si.smooth(.99);
    disperse(i) = wa.allpass2(i*100+30*wow*i%2000,1+60*dfactor,0)
                  : wa.allpass2(i*100-70*wow*i%2000,1+50*dfactor,0);
    outl = +~(de.delay(max_del, Dl-1) : *(feedback) : fi.lowpass(4,10000)
             : de.delay(400, 3000*wow) : par(i,16,disperse(i)) : aa.tanh1
             : fi.highpass(2,100));
    outr = +~(de.delay(max_del, Dr-1) : *(feedback) : fi.lowpass(4,10000)
             : de.delay(400, 1500*wow) : par(i,16,disperse(i)) : aa.tanh1
             : fi.highpass(2,100));
};

// EQ CHANNEL (from Main.dsp)
aa_eqchl(l, h, lp, hp, blf, blv, d, o) = outl, outr
with{
    L = (l-0.5)*12; H = (h-0.5)*12;
    Lp = 20*2^(10*lp); Hp = 20*2^(10*hp);
    Blf = 20*2^(10*blf); Blv = (blv-0.5)*6;
    preamp = hgroup("[0]pre",_,_:(ef.cubicnl_nodc(d,o):fi.dcblocker),(ef.cubicnl_nodc(d,o):fi.dcblocker));
    shelves = vgroup("[1]shelf",_,_:(fi.low_shelf(L,500):fi.high_shelf(H,1500)),
                   (fi.low_shelf(L,500):fi.high_shelf(H,1500)));
    passes = vgroup("[2]passes",_,_:(fi.resonlp(Lp,1,1):fi.resonhp(Hp,1,1)),
                  (fi.resonlp(Lp,1,1):fi.resonhp(Hp,1,1)));
    bell = vgroup("bl",_,_:fi.peak_eq(Blv,Blf,200),fi.peak_eq(Blv,Blf,200));
    strip = preamp:(shelves:passes:bell);
    outl, outr = strip;
};

// AD Envelope
aa_drum_ad(trig, a, d, ex_f) = out
with{
    attack_time = 10 + a*.500*ma.SR; decay_time = d*1.5*ma.SR;
    trg = trig <: _,_@attack_time : _+_ : _;
    env_time(counterpos) = ba.if(counterpos, attack_time, decay_time);
    count = ba.counter(trg) % 2; line = count : ba.line(env_time(count));
    ex = (ex_f-.5)*10 : round; exf(in) = pow(in, ba.if(ex>1,ex,(-1/ex)));
    out = line : exf;
};

       main_trig = (trig1 + trig2 + trig3 + trig4) : ba.impulsify;
       
       // Use frequency from lane 1 (or combine all)
       note = 60 + (f1 * 48); // 0-1 -> 60-108 (C4-C7)
       freq = 440.0 * ma.pow(2.0, (note-69.0)/12.0);
       
       // Simple subtractive synth
       saw = os.saw(freq);
       square = os.square(freq*1.5);
       osc = saw + square*0.3;
       
       // Envelope
       env = en.adsr(0.001, 0.1, 0.3, 0.1, main_trig);
       
       // Filter with control from c1
       cutoff = 200 + c1 * 10000;
       filtered = osc : fi.lowpass(cutoff, 0.5);
       
       // Apply velocity and envelope
       out = filtered * env * v1 * 0.5, filtered * env * v1 * 0.5;
   };

5. TEST THE FIXES
